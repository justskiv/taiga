{{- /* CSS bundle: generated palette vars (first) + the numbered theme sheets in name order (the cascade), minus any sheet whose feature is switched off, + the site's optional css/custom.css (last, so site overrides win without a fork). dev serves it raw; prod minifies and fingerprints. */ -}} {{- $palettes := partial "head/palettes.html" . | resources.FromString "css/00-palettes.css" -}} {{- $bundle := slice $palettes -}} {{- /* Optional sheets. A feature that is switched off must not leave its rules behind: "off" has to mean the bundle carries not one line of it, not that the selectors match nothing. The sheet keeps its number — and with it its place in the cascade; only the decision lives here. The target name carries the decision too. resources.Concat is keyed by target path alone, so a single fixed name would hand every language whichever bundle happened to be built first — and the flag is read per language, like everything else in the feature. */ -}} {{- $skip := slice -}} {{- $tag := "" -}} {{- if not (partialCached "newsletter/cfg.html" . .Language.Lang).enable -}} {{- $skip = $skip | append "28-newsletter.css" -}} {{- $tag = printf "%s-no-newsletter" $tag -}} {{- end -}} {{- range sort (resources.Match "css/[0-9]*.css") "Name" -}} {{- if in $skip (path.Base .Name) }}{{ continue }}{{ end -}} {{- $bundle = $bundle | append . -}} {{- end -}} {{- with resources.Get "css/custom.css" -}}{{ $bundle = $bundle | append . }}{{- end -}} {{- $css := $bundle | resources.Concat (printf "css/bundle%s.css" $tag) -}} {{- if hugo.IsProduction -}} {{- with $css | minify | fingerprint "sha256" }} {{- end -}} {{- else -}} {{- end -}} {{- /* page-bundle widget styles: one folder per widget under widgets/, plus a _shared/ level for classes several widgets use. Concatenated in name order (_shared sorts before w-*), minified + fingerprinted, linked only here. The Concat target MUST be page-unique: Hugo caches concatenated resources by target path, so a fixed name would collide across pages — every page would serve whichever bundle won the cache. Key it by the page's UniqueID. */ -}} {{- $wcss := sort (.Resources.Match "widgets/**.css") "Name" -}} {{- with $wcss -}} {{- $bundle := . | resources.Concat (printf "widgets/bundle-%s.css" $.File.UniqueID) -}} {{- if hugo.IsProduction -}} {{- with $bundle | minify | fingerprint "sha256" }} {{- end -}} {{- else -}} {{- end -}} {{- end -}}
