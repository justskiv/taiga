{{- /* Newsletter config, resolved once and read by everything else — the partials, the shortcodes and the js-bridge. Returns a dict, never nil, so a caller can say `$nl.enable` without guarding for a site that never heard of the feature. The theme's own hugo.toml carries the defaults (enable = false, the two chrome placements on, the two content placements off); Hugo merges a site's [params.newsletter] over them key by key. The `isset` dance below is belt-and-braces for the booleans whose default is TRUE: `| default` cannot tell "absent" from "deliberately false", and a header button that ignores `header = false` would be a bug the site owner cannot fix. The keys are looked up LOWERCASED: Hugo normalises every config key to lower case, and `isset`/`index` — unlike `.Params.someKey` — are plain map lookups, so `isset $pl "seriesLanding"` is false on a site that set exactly that key. The camel case survives in the returned dict, which is ours and is read by name everywhere else. Usage: {{ $nl := partialCached "newsletter/cfg.html" . }} */ -}} {{- $nl := site.Params.newsletter | default dict -}} {{- $pl := $nl.placements | default dict -}} {{- $flag := dict -}} {{- range $key, $default := dict "header" true "footer" true "articleEnd" false "seriesLanding" false -}} {{- $v := $default -}} {{- $k := lower $key -}} {{- if isset $pl $k }}{{ $v = index $pl $k }}{{ end -}} {{- $flag = merge $flag (dict $key $v) -}} {{- end -}} {{- $enable := false -}}{{ if isset $nl "enable" }}{{ $enable = $nl.enable }}{{ end -}} {{- $mode := $nl.mode | default "worker" -}} {{- if not (in (slice "worker" "direct") $mode) -}} {{- errorf "newsletter: params.newsletter.mode is %q — the only modes are \"worker\" (a proxy on your own origin) and \"direct\" (listmonk's public endpoint)" $mode -}} {{- end -}} {{- if and $enable (eq $mode "direct") (not $nl.listUUID) -}} {{- errorf "newsletter: mode = \"direct\" needs params.newsletter.listUUID — the public listmonk endpoint subscribes by list UUID, and without one every submission would fail silently" -}} {{- end -}} {{- /* The subscribe page is written as a site path ("/subscribe/") and has to survive both a subpath install and a second language. relLangURL does exactly that — but ONLY for an input without a leading slash: given one, it adds the language and silently drops the baseURL's own path, so a demo under /taiga/ would link to /ru/subscribe/ and 404. An absolute URL has no leading slash to trim and passes through untouched. */ -}} {{- $sub := $nl.subscribePage | default "/subscribe/" -}} {{- /* The address the failed form offers to write to. It falls back to the letter's own From address — on most sites they are the same mailbox, and a site that sends from a no-reply can say so by naming another one here. Empty on both ⇒ the hint line is never rendered: an offer to write to nobody is worse than no offer. */ -}} {{- $contact := $nl.contactEmail | default $nl.fromAddress | default "" -}} {{- return (dict "enable" $enable "mode" $mode "endpoint" ($nl.endpoint | default "/api/subscribe") "listUUID" ($nl.listUUID | default "") "turnstileSiteKey" ($nl.turnstileSiteKey | default "") "contactEmail" $contact "subscribePage" (strings.TrimPrefix "/" $sub | relLangURL) "placements" $flag ) -}}
